class Solution {
public:
    int majorityElement(vector<int>& nums) {
        map<int, int> m;
	    for (int i : nums) {
	        m[i]+=1;
	    }   
        int max = 0;
        int max_num = 0;
        for (auto i : m) {
            if (i.second > max) {
                max = i.second;
                max_num = i.first;
            }
        }

        return max_num;
    }
};
